74    fn from(scheme: MessageBoxColorScheme) -> [sys::SDL_MessageBoxColor; 5] {
75        fn to_message_box_color(t: (u8, u8, u8)) -> sys::SDL_MessageBoxColor {
76            sys::SDL_MessageBoxColor {
77                r: t.0,
78                g: t.1,
79                b: t.2,

81        }
82        [
83            to_message_box_color(scheme.background),
84            to_message_box_color(scheme.text),
85            to_message_box_color(scheme.button_border),
86            to_message_box_color(scheme.button_background),

89    }

107#[derive(Debug, Clone)]
108pub enum ShowMessageError {
109    InvalidTitle(NulError),
110    InvalidMessage(NulError),
111    /// Second argument of the tuple (i32) corresponds to the
112    /// first button_id having an error
113    InvalidButton(NulError, i32),
114    SdlError(String),
115}
116
117impl fmt::Display for ShowMessageError {
118    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
119        use self::ShowMessageError::*;
120
121        match *self {
122            InvalidTitle(ref e) => write!(f, "Invalid title: {}", e),
123            InvalidMessage(ref e) => write!(f, "Invalid message: {}", e),
124            InvalidButton(ref e, value) => write!(f, "Invalid button ({}): {}", value, e),
125            SdlError(ref e) => write!(f, "SDL error: {}", e),
126        }
127    }
128}
129
130impl error::Error for ShowMessageError {
131    fn source(&self) -> Option<&(dyn error::Error + 'static)> {
132        match self {
133            Self::InvalidTitle(err) => Some(err),
134            Self::InvalidMessage(err) => Some(err),
135            Self::InvalidButton(err, _) => Some(err),
136            Self::SdlError(_) => None,
137        }
138    }
139}

141/// Show a simple message box, meant to be informative only.
142///
143/// There is no way to know if the user clicked "Ok" or closed the message box,
144/// If you want to retrieve which button was clicked and customize a bit more
145/// your message box, use `show_message_box` instead.
146#[doc(alias = "SDL_ShowSimpleMessageBox")]
147pub fn show_simple_message_box<'a, W>(
148    flags: MessageBoxFlag,
149    title: &str,
150    message: &str,
151    window: W,
152) -> Result<(), ShowMessageError>
153where
154    W: Into<Option<&'a Window>>,
155{
156    use self::ShowMessageError::*;
157    let result = unsafe {
158        let title = match CString::new(title) {
159            Ok(s) => s,
160            Err(err) => return Err(InvalidTitle(err)),
161        };
162        let message = match CString::new(message) {
163            Ok(s) => s,
164            Err(err) => return Err(InvalidMessage(err)),
165        };
166        sys::SDL_ShowSimpleMessageBox(
167            flags.bits(),
168            title.as_ptr() as *const c_char,
169            message.as_ptr() as *const c_char,
170            window.into().map_or(ptr::null_mut(), |win| win.raw()),
171        )
172    } == 0;
173
174    if result {
175        Ok(())
176    } else {
177        Err(SdlError(get_error()))
178    }
179}

181/// Show a customizable message box.
182///
183/// An array of buttons is required for it to work. The array can be empty,
184/// but it will have no button beside the close button.
185///
186/// On success, it will return either the button clicked or the close button.
187/// Note that the variant of the `ClickedButton` enum will also be returned if the message box
188/// has been forcefully closed (Alt-F4, ...)
189///
190#[doc(alias = "SDL_ShowMessageBox")]
191pub fn show_message_box<'a, 'b, W, M>(
192    flags: MessageBoxFlag,
193    buttons: &'a [ButtonData],
194    title: &str,
195    message: &str,
196    window: W,
197    scheme: M,
198) -> Result<ClickedButton<'a>, ShowMessageError>
199where
200    W: Into<Option<&'b Window>>,
201    M: Into<Option<MessageBoxColorScheme>>,
202{
203    let window = window.into();
204    let scheme = scheme.into();
205
206    use self::ShowMessageError::*;
207    let mut button_id: c_int = 0;
208    let title = match CString::new(title) {
209        Ok(s) => s,
210        Err(err) => return Err(InvalidTitle(err)),
211    };
212    let message = match CString::new(message) {
213        Ok(s) => s,
214        Err(err) => return Err(InvalidMessage(err)),
215    };
216    let button_texts: Result<Vec<_>, (_, i32)> = buttons
217        .iter()
218        .map(|b| CString::new(b.text).map_err(|e| (e, b.button_id)))
219        .collect(); // Create CString for every button; and catch any CString Error
220    let button_texts = match button_texts {
221        Ok(b) => b,
222        Err(e) => return Err(InvalidButton(e.0, e.1)),
223    };
224    let raw_buttons: Vec<sys::SDL_MessageBoxButtonData> = buttons
225        .iter()
226        .zip(button_texts.iter())
227        .map(|(b, b_text)| sys::SDL_MessageBoxButtonData {
228            flags: b.flags.bits(),
229            buttonid: b.button_id as c_int,
230            text: b_text.as_ptr(),
231        })
232        .collect();
233    let result = unsafe {
234        let scheme = scheme.map(|scheme| sys::SDL_MessageBoxColorScheme {
235            colors: scheme.into(),
236        });
237        let msg_box_data = sys::SDL_MessageBoxData {
238            flags: flags.bits(),
239            window: window.map_or(ptr::null_mut(), |win| win.raw()),
240            title: title.as_ptr() as *const c_char,
241            message: message.as_ptr() as *const c_char,
242            numbuttons: raw_buttons.len() as c_int,
243            buttons: raw_buttons.as_ptr(),
244            colorScheme: scheme
245                .as_ref()
246                .map(|p| p as *const _)
247                .unwrap_or(ptr::null()),
248        };
249        sys::SDL_ShowMessageBox(&msg_box_data as *const _, &mut button_id as &mut _)
250    } == 0;
251    if result {
252        match button_id {
253            -1 => Ok(ClickedButton::CloseButton),
254            id => {
255                let button = buttons.iter().find(|b| b.button_id == id);
256                Ok(ClickedButton::CustomButton(button.unwrap()))
257            }
258        }
259    } else {
260        Err(SdlError(get_error()))
261    }
