510impl fmt::Display for Scancode {
511    fn fmt(&self, f: &mut fmt::Formatter) -> Result<(), fmt::Error> {
512        write!(f, "{}", self.name())
513    }

515
516use crate::keyboard::Keycode;
517
518impl Scancode {
519    /// Gets the scancode from a virtual key. Returns None if there is no corresponding scancode.
520    #[doc(alias = "SDL_GetScancodeFromKey")]
521    pub fn from_keycode(keycode: Keycode) -> Option<Scancode> {
522        unsafe {
523            match sys::SDL_GetScancodeFromKey(keycode.into()) {
524                SDL_Scancode::SDL_SCANCODE_UNKNOWN => None,
525                scancode_id => Scancode::from_i32(scancode_id as i32),
526            }
527        }
528    }
529
530    #[doc(alias = "SDL_GetScancodeFromName")]
531    pub fn from_name(name: &str) -> Option<Scancode> {
532        unsafe {
533            match CString::new(name) {
534                Ok(name) => match sys::SDL_GetScancodeFromName(name.as_ptr() as *const c_char) {
535                    SDL_Scancode::SDL_SCANCODE_UNKNOWN => None,
536                    scancode_id => Some(Scancode::from_i32(scancode_id as i32).unwrap()),
537                },
538                // string contains a nul byte - it won't match anything.
539                Err(_) => None,
540            }
541        }
542    }
543
544    #[doc(alias = "SDL_GetScancodeName")]
545    pub fn name(self) -> &'static str {
546        // The name string pointer lives in static, read-only memory.
547        // Knowing this, we can always return a string slice.
548        unsafe {
549            let buf = sys::SDL_GetScancodeName(transmute::<u32, SDL_Scancode>(self as u32));
550            ::std::str::from_utf8(CStr::from_ptr(buf as *const _).to_bytes()).unwrap()
551        }
552    }
553}
