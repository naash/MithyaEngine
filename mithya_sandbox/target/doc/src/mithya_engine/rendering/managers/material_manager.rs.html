13// Material Manager for caching and loading
14pub struct MaterialManager {
15    //Not sure if one manager should have direct reference to other managers. Reconsider this.
16    shader_manager: ShaderManager,
17    texture_manager: TextureManager,
18    //,,,,,,,,,,,,
19    materials: HashMap<u32, Material>,        // ID -> Material
20    material_name_to_id: HashMap<String, u32>,         // Name -> ID (for lookup by name)
21    next_id: u32,
22    shader_cache: HashMap<String, u32>, // Path -> Program ID
23}
24
25impl MaterialManager {
26    pub fn new() -> Self {
27        Self {
28            material_name_to_id: HashMap::new(),
29            materials: HashMap::new(),
30            next_id: 1,
31            shader_manager: ShaderManager::new(),
32            texture_manager: TextureManager::new(),
33            shader_cache: HashMap::new(),
34        }
35    }
36
37    // Create material with auto-generated ID
38    pub fn load_material(&mut self, name: &str, vertex_path: &str, fragment_path: &str) -> Result<(), String> {
39        let shader_key = format!("{}+{}", vertex_path, fragment_path);
40        
41        let id = self.next_id;
42        self.next_id += 1;
43
44        // Check if shader is already compiled
45        let program_key = if let Some(&existing_id) = self.shader_cache.get(&shader_key) {
46            existing_id
47        } else {
48            // Compile new shader program
49            let program_key = self.shader_manager.create_program(vertex_path, fragment_path)?;
50            self.shader_cache.insert(shader_key, program_key);
51            program_key
52        };
53
54        let mut material = Material::new(name, vertex_path, fragment_path);
55        let program_id =self.shader_manager.get_program(program_key);
56                
57        //This is ugly, needs cleaup
58        material.shader_program_id = Some(program_id.unwrap().id());
59        println!("Loading Material with Program : {}", &material.shader_program_id.unwrap());
60
61        self.materials.insert(id, material);
62        self.material_name_to_id.insert(name.to_string(), id);
63        Ok(())
64    }
65
66    pub fn get_material_id_from_name(&self, name: &str) -> Option<&u32> {
67        self.material_name_to_id.get(name)
68    }
69
70    pub fn get_material(&self, material_id: &u32) -> Option<&Material> {
71        self.materials.get(&material_id)
72    }
73
74    pub fn get_material_mut(&mut self, material_id: &u32) -> Option<&mut Material> {
75        self.materials.get_mut(&material_id)
76    }
77
78    pub fn get_material_from_name(&self, name: &str) -> Option<&Material> {
79        let material_id = self.material_name_to_id[name];
80        self.materials.get(&material_id)
81    }
82
83    pub fn get_material_from_name_mut(&mut self, name: &str) -> Option<&mut Material> {
84        let material_id = self.material_name_to_id[name];
85        self.materials.get_mut(&material_id)
86    }
87
88    pub fn add_texture_to_material(
89    &mut self,
90    material_name: &str,
91    uniform_name: &str,
92    texture_path: &str,
93    slot: u32,
94    ) -> Result<(), MaterialError> {
95
96        // Load texture through texture manager
97        let texture_id = self.texture_manager.load_texture(texture_path)
98            .map_err(MaterialError::TextureLoadError)?;
99
100        // Add texture binding to material
101        let material_id = self.material_name_to_id.get(material_name);
102        if let Some(material) = self.materials.get_mut(material_id.unwrap()) {
103            material.bind_texture(uniform_name, texture_id, slot);
104            Ok(())
105        } else {
106            Err(MaterialError::MaterialNotFound(material_name.to_string()))
107        }
108    }
109
110    // Create predefined materials
111    pub fn create_default_materials(&mut self) -> Result<(), String> {
112        self.load_material("unlit_color",
113         include_str!("../../../shaders/unlit_color.vert"),
114         include_str!("../../../shaders/unlit_color.frag"))?;
115
116        self.load_material("unlit_texture_default",
117         include_str!("../../../shaders/unlit_texture.vert"),
118         include_str!("../../../shaders/unlit_texture.frag"))?;
119
120        self.load_material("unlit_texture_circle",
121         include_str!("../../../shaders/unlit_texture.vert"),
122         include_str!("../../../shaders/unlit_texture.frag"))?;
123
124        //Load default texture for this material
125        let _ = self.add_texture_to_material("unlit_texture_default",
126             "u_texture",
127              "test_texture.png",
128              0);
129
130        let _ = self.add_texture_to_material("unlit_texture_circle",
131        "u_texture",
132        "pinkCircle.png",
133        1);
134
135         Ok(())     
136    }
137}
