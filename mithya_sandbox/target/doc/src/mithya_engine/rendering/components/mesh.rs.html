6#[derive(Clone, Debug)]
7pub struct VertexAttribute {
8    pub location: u32,
9    pub size: i32,        // Number of components (1, 2, 3, or 4)
10    pub offset: usize,    // Offset in bytes from start of vertex
11}
12
13// Mesh data - the actual geometry
14#[derive(Clone, Debug)]
15pub struct Mesh {
16    pub vertices: Vec<f32>,
17    pub indices: Vec<u32>,
18    pub vao: Option<gl::types::GLuint>,
19    pub vbo: Option<gl::types::GLuint>,
20    pub ebo: Option<gl::types::GLuint>,
21    pub vertex_stride: usize,           // Size of one vertex in bytes
22    pub attributes: Vec<VertexAttribute>, // Flexible attribute definition
23}
24
25impl Mesh {
26    pub fn new_triangle() -> Self {
27        Self {
28            vertices: vec![-5.0, -5.0, 0.0, 5.0, -5.0, 0.0, 0.0, 0.5, 0.0],
29            indices: vec![0, 1, 2],
30            vao: None,
31            vbo: None,
32            ebo: None,
33            vertex_stride: 3 * std::mem::size_of::<f32>(),
34            attributes: vec![
35                VertexAttribute { location: 0, size: 3, offset: 0 }, // Position only
36            ],
37        }
38    }
39
40    pub fn new_quad() -> Self {
41        Self {
42            vertices: vec![
43                -0.5, -0.5, 0.0,  // Bottom left
44                 0.5, -0.5, 0.0,  // Bottom right
45                 0.5,  0.5, 0.0,  // Top right
46                -0.5,  0.5, 0.0,  // Top left
47            ],
48            indices: vec![0, 1, 2, 2, 3, 0],
49            vao: None,
50            vbo: None,
51            ebo: None,
52            vertex_stride: 3 * std::mem::size_of::<f32>(),
53            attributes: vec![
54                VertexAttribute { location: 0, size: 3, offset: 0 }, // Position only
55            ],
56        }
57    }
58
59    // Textured versions
60    pub fn new_triangle_textured() -> Self {
61        Self {
62            vertices: vec![
63                // Position   UV
64                -0.5, -0.5, 0.0,  0.0, 0.0,  // Bottom left
65                 0.5, -0.5, 0.0,  1.0, 0.0,  // Bottom right
66                 0.0,  0.5, 0.0,  0.5, 1.0,  // Top center
67            ],
68            indices: vec![0, 1, 2],
69            vao: None,
70            vbo: None,
71            ebo: None,
72            vertex_stride: 5 * std::mem::size_of::<f32>(),
73            attributes: vec![
74                VertexAttribute { location: 0, size: 3, offset: 0 }, // Position
75                VertexAttribute { location: 1, size: 2, offset: 3 * std::mem::size_of::<f32>() }, // UV
76            ],
77        }
78    }
79
80    pub fn new_quad_textured() -> Self {
81        Self {
82            vertices: vec![
83                // Position    UV
84                -0.5, -0.5, 0.0,  0.0, 0.0,  // Bottom left
85                 0.5, -0.5, 0.0,  1.0, 0.0,  // Bottom right
86                 0.5,  0.5, 0.0,  1.0, 1.0,  // Top right
87                -0.5,  0.5, 0.0,  0.0, 1.0,  // Top left
88            ],
89            indices: vec![0, 1, 2, 2, 3, 0],
90            vao: None,
91            vbo: None,
92            ebo: None,
93            vertex_stride: 5 * std::mem::size_of::<f32>(),
94            attributes: vec![
95                VertexAttribute { location: 0, size: 3, offset: 0 }, // Position
96                VertexAttribute { location: 1, size: 2, offset: 3 * std::mem::size_of::<f32>() }, // UV
97            ],
98        }
99    }
